```python
# 层1：飞行状况    auto_flight_state ∈ {'flying', 'landed'}       # 接收机判定
# 层2：机构状况    auto_mech_state  ∈ {'idle', 'hold', 'process'}  # 队列判定
```

## 二、线程装配（生产者，start 时启动）

```python
# ── STM32 接收线程 / 相机取帧线程：照旧（actuator_state / current_frame）──
# ── IMU 落地监视线程：并列备选源，去留待定 ★ ──

def 接收机读数线程():                        # 数据源待电控接入，先占位
    while 运行:
        if 收到遥控器数据():
            data.receiver_last_time = now()
        sleep(短间隔)

def 机构监视线程():                           # 重写：长队列判三态，低频采样
    队列 = deque(maxlen=params.MECH_QUEUE_LEN)
    while 运行:
        样本 = judge_grabber(data.actuator_state)   # 'idle'/'cube'/'ball'/None
        队列.append(样本)
        data.auto_mech_state = 队列判三态(队列, data.auto_mech_state)
        sleep(params.MECH_SAMPLE_INTERVAL)          # 低频：静态数据靠数据量判定

def 队列判三态(队列, 当前):
    有效 = [s for s in 队列 if s is not None]       # ★运动中样本(None)先按丢弃
    if len(有效) < params.MECH_WINDOW:
        return 当前                                  # 样本不够 → 维持原状态
    近段 = 有效[-params.MECH_WINDOW:]
    if all(s in ('cube', 'ball') for s in 近段):
        return 'hold'                                # 近段连续抓着东西
    if all(s == 'idle' for s in 近段):
        if any(s in ('cube', 'ball') for s in 有效[:-params.MECH_WINDOW]):
            return 'process'   # 先抓着、后空了 → 转变态（长队列的意义）
                               # ★老样本滑出队列 → 自然退回 idle，即过程窗长=队列长
        return 'idle'                                # 一直是空爪常态
    return 当前                                      # ★混合段(正在动)→维持，待定
```

## 三、while 大循环（消费者/调度，全量逻辑）

```python
while 流程运行:
    # ═══ 层1 判定：飞行状况（每轮先刷，主源=接收机）═══
    # 飞行中人工持续操作 → 接收机有数据；落地后基本不再操作 → 超时无数据
    if data.receiver_last_time is None or \
            now() - data.receiver_last_time > params.RECEIVER_LOST_TIMEOUT:
        data.auto_flight_state = 'landed'   # ★开机未飞也恒判 landed（在地上），符合预期?
    else:
        data.auto_flight_state = 'flying'
    # 并列备选（换源只换这段，骨架不动；注释并列关系）：
    #   备选1 IMU：落地冲击尖峰触发 → landed（监视线程已有 landing_imu_hit）
    #   备选2 光流：连续帧画面几乎不动 → landed（check_frame_still）

    # ═══ 完赛判定 ═══
    if cube_done >= 15:
        提示人工"任务完成"                    # ★G1 收尾形态：提示后停循环?
        break

    品类 = 本轮品类()

    # ═══════════ 第一层 if：飞行状况 ═══════════
    if data.auto_flight_state == 'flying':

        # ─────── 第二层 if：机构状况 ───────
        if data.auto_mech_state == 'idle':
            # 情况1 空爪飞行（初始待飞 / 投放完成后飞回途中）
            pass                              # ★无动作，等落地

        elif data.auto_mech_state == 'hold':
            # 情况2 抓着东西转场飞行
            pass                              # ★无动作，等落地

        elif data.auto_mech_state == 'process':
            # 情况3 飞行中掉落（空中 抓着→空）
            # 叠加投放完成判定：空中松开 = 掉落，不算投放完成（★旧决议待重申）
            if data.auto_grabbed is not None and not data.process_handled:
                提示人工"飞行中掉落，重新飞回抓取"
                data.auto_grabbed = None      # 清标记，N 不变
                data.process_handled = True   # ★一次性处理；process 退回 idle 时复位

    elif data.auto_flight_state == 'landed':

        # ─────── 第二层 if：机构状况 ───────
        if data.auto_mech_state == 'idle':

            # ─── 第三层 if：抓取任务进度记账 ───
            if data.task_memo == 'grab_sent':
                # 抓取指令已发 → 等验证（机构一夹住翻到 hold，在 hold 侧收口）
                if now() - data.task_memo_time > params.GRAB_VERIFY_TIMEOUT:
                    data.grab_fail_cnt += 1
                    data.task_memo = None     # 回 YOLO 重新选靶
                    if data.grab_fail_cnt >= 3:          # ★3 次上限（旧决议）
                        提示人工"连续抓取失败，重新飞"    # N 不变，等下一次落地
            else:
                # ─── 第三层 if：YOLO 任务判定 ───
                帧 = data.current_frame
                if 帧 is None:
                    提示"无画面，等待"                # ★相机断线只提示?
                    continue
                目标 = 识别防抖(帧, 只看=品类)         # 连续M帧一致（沿用旧阶段二）
                if 目标.本轮品类圈内非空:
                    # 情况4a 可抓
                    靶 = 评分选靶(圈内目标, 帧, 品类)[0]
                    发抓取指令(像素→现实坐标(靶.中点)) # common 协议/坐标映射
                    data.task_memo = 'grab_sent'
                    data.task_memo_time = now()
                else:
                    # 情况4b 不可抓
                    提示人工"无本轮品类目标，重新飞"   # ★每轮重发提示?防重发待定

        elif data.auto_mech_state == 'hold':

            # ─── 第三层 if：抓取成功收口 / 投放任务进度 ───
            if data.task_memo == 'grab_sent':
                # 指令已发且机构已夹住 → 抓取成功
                data.auto_grabbed = 品类
                data.task_memo = None
                data.grab_fail_cnt = 0
                # 下一轮走 landed+hold → 投放链路（人工起飞后同理）
            elif data.task_memo == 'drop_sent':
                pass        # 等机构松开（层2→process 由情况6收尾）；★超时提示待定
            else:
                # ─── 第三层 if：YOLO 任务判定 ───
                帧 = data.current_frame
                if 帧 is None:
                    提示"无画面，等待"
                    continue
                容器 = 识别防抖(帧, 只看=对应容器)     # 魔方→盆 / 球→桶
                if 容器可投(容器):                     # 盆=重合面积≥阈值 / 桶=中点在圈
                    # 情况5a 可投
                    发投放指令()                        # common 协议
                    data.task_memo = 'drop_sent'
                    data.task_memo_time = now()
                else:
                    # 情况5b 不可投
                    提示人工"不可投，重新飞"            # ★N不变账保持（旧口径）

        elif data.auto_mech_state == 'process':
            # 情况6 落地松开中（主动投放松开 / 落地自发掉落 都到这）
            # 叠加投放完成判定：落地时 抓着→空 = 投放完成（★旧决议：都算完成）
            if data.auto_grabbed is not None and not data.process_handled:
                data.N += 1
                cube_done/ball_done 对应 += 1
                data.auto_grabbed = None
                data.task_memo = None
                data.process_handled = True   # ★一次性处理；process 退回 idle 时复位
                提示人工"投放完成，飞回抓取区"

    sleep(params.FLOW_LOOP_INTERVAL)


def 本轮品类():
    # ★N 规则照旧：第1次魔方，奇魔偶球；球尽(或N≥12)后全魔方；魔方15完赛
    if cube_done >= 15:  return '完赛'
    if ball_done >= 6 or N >= 12:  return 'cube'
    return 'cube' if N % 2 == 1 else 'ball'
```

## 四、★ 待你改/拍板的点（对应伪代码内标记）

1. 层2 队列判定：None（运动中）样本按丢弃处理对不对；混合段维持原状态
   对不对；process 窗长 = 队列长（老样本滑出自然退回 idle）这个机制行不行。
2. 层1 主判据口径："收到数据"=链路在线还是通道值有变化；开机未飞恒判
   landed 是否符合预期。
3. 情况1/2 飞行中无动作只等落地——确认。
4. 情况3 飞行中掉落不算投放完成、清标记 N 不变——旧决议重申一下。
5. 情况4b/5b 提示人工的防重发（每轮都提示还是提示一次挂标记）。
6. 抓取验证超时（GRAB_VERIFY_TIMEOUT）与 3 次失败上限——数值与口径。
7. 投放验证只等 process 出现，要不要单独超时（drop_sent 超时提示）。
8. 情况6 的守卫条件：auto_grabbed 非空才记账（防空爪误判）——确认。
9. process_handled 一次性标记的复位时机（process 退回 idle 时由谁复位）。
10. 完赛形态：提示后 break 停循环还是停在提示。
11. IMU 监视线程去留（并列备选保留 or 停用）。
